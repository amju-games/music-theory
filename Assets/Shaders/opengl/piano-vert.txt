// Title piano vert shader

attribute vec4 position;
attribute vec2 uv;
varying vec2 vUV;
out vec4 clipSpacePos;
uniform mat4 modelViewProjectionMatrix;

void main()
{
  // Use the screen space position as UV to give fake reflection effect...
  //  Not a great effect, would be better if piano rotating?
  clipSpacePos = modelViewProjectionMatrix * position;
  gl_Position = clipSpacePos;

  vUV = uv;
}
