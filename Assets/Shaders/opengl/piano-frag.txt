// Title piano frag shader

uniform vec4 colourUniform;
uniform sampler2D tex;

varying vec2 vUV;
varying vec2 vUVReflect;

void main() 
{    
    // Look up reflection colour (sphere mapping)
    vec3 reflectionColor = texture2D(tex, vUVReflect).rgb;

    // Look up diffuse map -- on the same texture.
    vec3 diffuseColour = texture2D(tex, vUV).rgb;

    vec3 col = diffuseColour;
    // Apply global colour to the piano -- but not the keys, which are
    //  on the right of .37 in x.
    col *= (vUV.x > 0.37 ? vec3(1.0, 1.0, 1.0) : colourUniform.rgb);

    // Add a bit of reflection colour: doesn't look great tbh.
    col += reflectionColor * 0.1;

    gl_FragColor = vec4(col, 1.0);
}
